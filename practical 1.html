#include int main() { int a[100], n, i, pos, value, search; printf("Enter number of elements: "); scanf("%d", &n); printf("Enter array elements: "); for(i = 0; i < n; i++) { scanf("%d", &a[i]); } printf("Array elements are: "); for(i = 0; i < n; i++) { printf("%d ", a[i]); } // Insertion printf("\n\nEnter position for insertion: "); scanf("%d", &pos); printf("Enter value to insert: "); scanf("%d", &value); for(i = n; i >= pos; i--) { a[i] = a[i - 1]; } a[pos - 1] = value; n++; printf("Array after insertion: "); for(i = 0; i < n; i++) { printf("%d ", a[i]); } // Deletion printf("\n\nEnter position for deletion: "); scanf("%d", &pos); for(i = pos - 1; i < n - 1; i++) { a[i] = a[i + 1]; } n--; printf("Array after deletion: "); for(i = 0; i < n; i++) { printf("%d ", a[i]); } // Searching printf("\n\nEnter element to search: "); scanf("%d", &search); for(i = 0; i < n; i++) { if(a[i] == search) { printf("Element found at position: %d\n", i + 1); return 0; } } printf("Element not found.\n"); return 0; }
